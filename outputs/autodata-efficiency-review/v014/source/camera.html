  20  def update_mounted_camera_poses(scene: InteractiveScene) -> None:
  21      """Refresh mounted camera poses before rendering or reading observations.
  22  
  23      The installed FrameView path can retain a stale world pose for a camera
  24      nested under an articulation. Use the public Camera API and live link state
  25      to preserve the calibrated mounting transform. No asset or dependency edits.
  26  
  27      Args:
  28          scene: OpenArm scene with the robot and its three camera sensors.
  29      """
  30      robot = scene["robot"]
  31      for name in CAMERA_NAMES:
  32          camera = scene[name]
  33          assert camera.cfg.offset.convention == "ros"
  34          parent_name = camera.cfg.prim_path.rsplit("/", 2)[-2]
  35          parent_pose = robot.data.body_link_pose_w.torch[:, robot.body_names.index(parent_name)]
  36          # Isaac Lab requires equal batch shapes; the mount is shared by all environments.
  37          position, rotation = combine_frame_transforms(
  38              parent_pose[:, :3],
  39              parent_pose[:, 3:],
  40              parent_pose.new_tensor(camera.cfg.offset.pos).expand_as(parent_pose[:, :3]),
  41              parent_pose.new_tensor(camera.cfg.offset.rot).expand_as(parent_pose[:, 3:]),
  42          )
  43          camera.set_world_poses(position, rotation, convention="ros")

  46  def rgb_image(env: ManagerBasedRLEnv, name: str) -> torch.Tensor:
  47      """Return a CPU uint8 RGB snapshot for the non-concatenated policy group.
  48  
  49      Cameras render on the GPU. The official recorder clones observations on
  50      their existing device and retains every frame until reset/export, so RGB
  51      observations must live on the CPU to avoid per-frame CUDA accumulation.
  52      Low-dimensional robot observations and simulation remain on the GPU.
  53      """
  54      update_mounted_camera_poses(env.scene)
  55      return env.scene[name].data.output["rgb"].torch[..., :3].to(device="cpu", copy=True)
